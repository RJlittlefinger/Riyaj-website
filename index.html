import os import numpy as np import pandas as pd from dotenv import load_dotenv # ১. পরিবেশ তৈরি ও ডেটা লোড load_dotenv() def calculate_legendary_indicators(df): """ মুভিং এভারেজ, MACD এবং ভলিউম ইন্ডিকেটর হিসাব করার ফাংশন """ # ২০ দিনের সিম্পল মুভিং এভারেজ (Trend Filter) df['SMA_20'] = df['close'].rolling(window=20).mean() # MACD গণনা (12, 26, 9) exp1 = df['close'].ewm(span=12, adjust=False).mean() exp2 = df['close'].ewm(span=26, adjust=False).mean() df['MACD'] = exp1 - exp2 df['Signal_Line'] = df['MACD'].ewm(span=9, adjust=False).mean() # ভলিউম মুভিং এভারেজ (স্মার্ট মানি ট্র্যাকিং) df['Vol_SMA'] = df['volume'].rolling(window=20).mean() return df def detect_candlestick_patterns(df): """ লেজেন্ডারি ক্যান্ডেলস্টিক এবং প্যানিক রিভার্সাল প্যাটার্ন খোঁজার লজিক """ # ক্যান্ডেলের বিভিন্ন অংশ নির্ধারণ df['Body'] = abs(df['close'] - df['open']) df['Total_Range'] = df['high'] - df['low'] df['Lower_Shadow'] = np.where(df['close'] > df['open'], df['open'] - df['low'], df['close'] - df['low']) df['Upper_Shadow'] = np.where(df['close'] > df['open'], df['high'] - df['close'], df['high'] - df['open']) # ১. হ্যামার ক্যান্ডেল (বুলিশ রিভার্সাল) # বডির তুলনায় নিচের শ্যাডো অন্তত দ্বিগুণ হবে এবং উপরের শ্যাডো খুব ছোট হবে df['Hammer'] = (df['Lower_Shadow'] >= 2 * df['Body']) & (df['Upper_Shadow'] <= 0.1 * df['Total_Range']) # ২. বুলিশ এনগালফিং (Bullish Engulfing) # আগের ক্যান্ডেল লাল এবং বর্তমান ক্যান্ডেল সবুজ যা আগেরটিকে পুরোপুরি ঢেকে ফেলে df['Bullish_Engulfing'] = (df['close'].shift(1) < df['open'].shift(1)) & \ (df['close'] > df['open']) & \ (df['open'] <= df['close'].shift(1)) & \ (df['close'] >= df['open'].shift(1)) # ৩. আপনার স্পেশাল 'প্যানিক রিভার্সাল' (Panic Drop Recovery) # টানা ৩টি ক্যান্ডেল লাল (প্যানিক সেল) এবং ভলিউম গড়ের চেয়ে বেশি, তারপরের ক্যান্ডেল সবুজ বা বাউন্স df['Consecutive_Red'] = (df['close'] < df['open']).rolling(window=3).sum() df['Panic_Drop'] = (df['Consecutive_Red'] == 3) & (df['volume'].shift(1) > df['Vol_SMA'].shift(1)) return df def generate_zero_loss_signals(df): """ সব নিয়ম মেনে নিখুঁত এন্ট্রি, টার্গেট এবং নো-লস স্টপ লস জেনারেট করার ফাংশন """ signals = [] for i in range(len(df)): if i < 20: # ইন্ডিকেটর সেটআপের জন্য প্রথম ২০টি ক্যান্ডেল বাদ continue # ক্যান্ডেলস্টিক কনফার্মেশন is_hammer = df['Hammer'].iloc[i] is_engulfing = df['Bullish_Engulfing'].iloc[i] is_panic_reversal = df['Panic_Drop'].iloc[i-1] and (df['close'].iloc[i] > df['open'].iloc[i]) # টেকনিক্যাল ও স্মার্ট মানি কনফার্মেশন is_macd_bullish = df['MACD'].iloc[i] > df['Signal_Line'].iloc[i] is_smart_volume = df['volume'].iloc[i] > df['Vol_SMA'].iloc[i] is_above_trend = df['close'].iloc[i] >= df['SMA_20'].iloc[i] # যদি ক্যান্ডেলস্টিক প্যাটার্ন মেলে এবং স্মার্ট মানি + MACD সিগন্যাল দেয় if (is_hammer or is_engulfing or is_panic_reversal) and is_macd_bullish and is_smart_volume: entry_price = df['close'].iloc[i] # আল্ট্রা-সেফ স্টপ লস: গত ৩টি ক্যান্ডেলের মধ্যে সবচেয়ে লো-এর চেয়েও কিছুটা নিচে (যাতে হিট না করে) recent_low = min(df['low'].iloc[i-2:i+1]) buffer = df['Total_Range'].iloc[i] * 0.15 # ১৫% এক্সট্রা সেফটি বাফার stop_loss = recent_low -
